\section{Polynomial Basis Conventions}
% ============================================================

\subsection{$P_1$ Polynomial}

$P_1$ uses a constrained form satisfying $P_1(0) = 0$, $P_1(1) = 1$:
\begin{equation}
P_1(x) = x + x(1-x) \cdot \tilde{P}_1(1-x)
\end{equation}
where $\tilde{P}_1$ is in powers of $(1-x)$:
\begin{equation}
\tilde{P}_1(y) = c_0 + c_1 y + c_2 y^2 + c_3 y^3
\end{equation}

\textbf{Admissibility:} Automatic from parameterization.

\subsection{$P_\ell$ Polynomials ($\ell \geq 2$)}

$P_\ell$ satisfies $P_\ell(0) = 0$:
\begin{equation}
P_\ell(x) = x \cdot \tilde{P}_\ell(x)
\end{equation}
where $\tilde{P}_\ell$ is in \textbf{monomial} basis:
\begin{equation}
\tilde{P}_\ell(x) = c_0 + c_1 x + c_2 x^2 + \cdots
\end{equation}

\textbf{Important:} Unlike $P_1$, the tilde coefficients for $P_2$ and $P_3$ use the standard monomial basis (powers of $x$), not powers of $(1-x)$.

\textbf{Admissibility:} Automatic from factor of $x$.

\subsection{$Q$ Polynomial}

$Q$ can be stored in two equivalent representations:

\textbf{Monomial basis:}
\begin{equation}
Q(x) = q_0 + q_1 x + q_2 x^2 + \cdots + q_5 x^5
\end{equation}

\textbf{$(1-2x)^k$ basis (PRZZ form):}
\begin{equation}
Q(x) = \sum_{k \in \{1,3,5\}} c_k (1-2x)^k
\end{equation}
with $Q(0) = 1$ constraint (automatically satisfied for odd $k$ only).

\textbf{Admissibility:} $Q(0) = 1$ enforced by basis choice.

% ============================================================
\section{Conversion Notes}
% ============================================================

\subsection{$(1-2t)^k$ Basis to Monomial}

For the Q polynomial, the PRZZ $(1-2t)^k$ basis coefficients convert to monomial form via:
\begin{align*}
Q(t) &= c_1(1-2t) + c_3(1-2t)^3 + c_5(1-2t)^5 \\
     &= [c_1 + c_3 + c_5] + [-2c_1 - 6c_3 - 10c_5]t + \cdots
\end{align*}

The monomial form is used directly in the production code for efficiency.

\subsection{Polynomial Evaluation Functions}

\textbf{$P_1$ evaluation:}
\begin{verbatim}
def eval_P1(x, tilde_coeffs):
    z = 1 - x
    p_tilde = sum(c * z**i for i, c in enumerate(tilde_coeffs))
    return x + x * z * p_tilde
\end{verbatim}

\textbf{$P_2$, $P_3$ evaluation:}
\begin{verbatim}
def eval_Pn(x, tilde_coeffs):
    p_tilde = sum(c * x**i for i, c in enumerate(tilde_coeffs))
    return x * p_tilde
\end{verbatim}
